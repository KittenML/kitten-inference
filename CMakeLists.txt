cmake_minimum_required(VERSION 3.18)
project(kitten_inference LANGUAGES C CXX)
if(APPLE)
    # Objective-C (ggml's Metal backend, engines/asr-cpu) and Objective-C++ (the TTS 0.8 Metal backend) for every
    # subdirectory: enabled only inside one subdirectory, the other's targets fail at generate time
    # ("CMAKE_OBJCXX_COMPILE_OBJECT" not set)
    enable_language(OBJC OBJCXX)
endif()
add_subdirectory(engines/lite/engine)
# Speech recognition on the CPU (and the Mac GPU): libkasr on llama.cpp's libraries, in kitten_inference/_asr_cpu/lib
set(_kasr_default ON)
if(WIN32)
    set(_kasr_arch "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
    if(NOT _kasr_arch)
        set(_kasr_arch "${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    string(TOLOWER "${_kasr_arch}" _kasr_arch)
    if(NOT _kasr_arch MATCHES "^(x64|amd64|x86_64)$")
        set(_kasr_default OFF)
    endif()
endif()
option(KITTEN_BUILD_ASR_CPU "Build the speech recognition CPU engine (engines/asr-cpu)" ${_kasr_default})
if(KITTEN_BUILD_ASR_CPU)
    add_subdirectory(engines/asr-cpu)
else()
    add_custom_target(kasr)  # pyproject.toml builds the target "kasr"; without the engine it is empty
endif()
install(DIRECTORY engines/tts2/tools/kt2n_assets/ DESTINATION kitten_inference/tts2/_assets COMPONENT python PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES engines/tts2/NOTICE DESTINATION kitten_inference/tts2 COMPONENT python)
install(DIRECTORY engines/tts2/licenses DESTINATION kitten_inference/tts2 COMPONENT python)
